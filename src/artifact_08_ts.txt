import express from "express";
import { createHash } from "node:crypto";
import { Connection, PublicKey } from "@solana/web3.js";

const RPC_URL = process.env.RPC_URL ?? "https://api.devnet.solana.com";
const PROGRAM_ID = new PublicKey(
  process.env.PROGRAM_ID ?? "PPuLsE1111111111111111111111111111111111111"
);
const POLL_MS = Number(process.env.POLL_MS ?? 5000);

const connection = new Connection(RPC_URL, "confirmed");
const seen = new Set<string>();
const app = express();

let lastPoll = 0;
let processed = 0;
let failed = 0;

function hashEvidence(input: Buffer | string): string {
  return createHash("sha256").update(input).digest("hex");
}

async function postJson(url: string, body: unknown): Promise<void> {
  const response = await fetch(url, {
    method: "POST",
    headers: { "content-type": "application/json" },
    body: JSON.stringify(body)
  });

  if (!response.ok) {
    throw new Error(`HTTP ${response.status} from ${url}`);
  }
}

async function notify(event: Record<string, unknown>): Promise<void> {
  const text =
    `ProofPulse milestone ${event.milestone} is ${event.status}. ` +
    `Evidence hash: ${event.evidenceHash}`;

  const jobs: Promise<void>[] = [];

  if (process.env.SLACK_WEBHOOK_URL) {
    jobs.push(postJson(process.env.SLACK_WEBHOOK_URL, { text }));
  }

  if (process.env.DISCORD_WEBHOOK_URL) {
    jobs.push(postJson(process.env.DISCORD_WEBHOOK_URL, { content: text }));
  }

  if (process.env.ENTERPRISE_WEBHOOK_URL) {
    jobs.push(postJson(process.env.ENTERPRISE_WEBHOOK_URL, event));
  }

  if (process.env.RESEND_API_KEY && process.env.NOTIFY_EMAIL) {
    jobs.push(
      postJson("https://api.resend.com/emails", {
        from: process.env.RESEND_FROM ?? "ProofPulse <noreply@example.com>",
        to: [process.env.NOTIFY_EMAIL],
        subject: `ProofPulse milestone ${event.status}`,
        text
      }).catch(async () => {
        const response = await fetch("https://api.resend.com/emails", {
          method: "POST",
          headers: {
            authorization: `Bearer ${process.env.RESEND_API_KEY}`,
            "content-type": "application/json"
          },
          body: JSON.stringify({
            from: process.env.RESEND_FROM,
            to: [process.env.NOTIFY_EMAIL],
            subject: `ProofPulse milestone ${event.status}`,
            text
          })
        });

        if (!response.ok) {
          throw new Error(`Resend failed with HTTP ${response.status}`);
        }
      })
    );
  }

  await Promise.all(jobs);
}

function decodeStatus(data: Buffer): string | null {
  // Anchor account data begins with an 8-byte discriminator.
  // This compact MVP reads the final status byte before the bump byte.
  if (data.length < 8 + 32 + 32 + 32 + 8 + 4 + 8 + 8 + 8 + 32 + 1 + 1) {
    return null;
  }

  const statusOffset = 8 + 32 + 32 + 32 + 8 + 4 + 128 + 8 + 8 + 8 + 32;
  const status = data[statusOffset];

  return ["Open", "Submitted", "Approved", "Rejected"][status] ?? null;
}

async function poll(): Promise<void> {
  lastPoll = Date.now();

  const accounts = await connection.getProgramAccounts(PROGRAM_ID, {
    commitment: "confirmed"
  });

  for (const account of accounts) {
    const status = decodeStatus(account.account.data);

    if (status !== "Approved" && status !== "Rejected") continue;

    const eventKey = `${account.pubkey.toBase58()}:${status}`;

    if (seen.has(eventKey)) continue;

    try {
      await notify({
        source: "solana",
        program: PROGRAM_ID.toBase58(),
        milestone: account.pubkey.toBase58(),
        status,
        evidenceHash: "available-from-account-decoder",
        observedAt: new Date().toISOString()
      });

      seen.add(eventKey);
      processed++;
    } catch (error) {
      failed++;
      console.error("notification failure", error);
    }
  }
}

app.get("/healthz", (_req, res) => {
  const healthy = Date.now() - lastPoll < POLL_MS * 5;
  res.status(healthy ? 200 : 503).json({
    ok: healthy,
    lastPoll,
    processed,
    failed,
    rpc: RPC_URL,
    program: PROGRAM_ID.toBase58()
  });
});

app.get("/readyz", (_req, res) => {
  res.status(lastPoll > 0 ? 200 : 503).json({ ready: lastPoll > 0 });
});

app.listen(Number(process.env.PORT ?? 3000), () => {
  console.log("ProofPulse worker listening");
});

setInterval(() => {
  poll().catch((error) => {
    failed++;
    console.error("poll failure", error);
  });
}, POLL_MS);

void poll();

export { hashEvidence, decodeStatus };
